```rust
pub fn link_entry_to_event(
    &self,
    entry_id: i64,
    event_uid: &str,
    event_snapshot: &str,  // JSON with summary, time, feed_color for orphan display
) -> Result<()> {
    self.conn().execute(
        r#"UPDATE diary_entries 
           SET event_uid = ?1, event_snapshot = ?2, updated_at = datetime('now')
           WHERE id = ?3"#,
        params![event_uid, event_snapshot, entry_id],
    )?;
    Ok(())
}
```

2. Add unlink_entry method:
```rust
/// Remove link between diary entry and calendar event
pub fn unlink_entry(&self, entry_id: i64) -> Result<()> {
    self.conn().execute(
        r#"UPDATE diary_entries 
           SET event_uid = NULL, updated_at = datetime('now')
           WHERE id = ?1"#,
        [entry_id],
    )?;
    Ok(())
}
```

Note: event_snapshot is kept on unlink so user can see what was linked.
