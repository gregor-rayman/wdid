```rust
CalendarResult::FeedData { feed_url, data, feed_name, feed_color } => {
    // Calculate date range (current date +/- 7 days for buffer)
    let current = self.view_state.current_date;
    let range_start = current - chrono::Duration::days(7);
    let range_end = current + chrono::Duration::days(7);
    
    // Parse events
    let events = parse_ical(
        &data,
        &feed_url,
        feed_name.as_deref(),
        feed_color.as_deref(),
        range_start,
        range_end,
    );
    
    // Clear old events for this feed and save new ones
    if let Err(e) = self.db.clear_feed_events(&feed_url) {
        eprintln!("Failed to clear feed events: {}", e);
    }
    if let Err(e) = self.db.save_events(&events) {
        eprintln!("Failed to save events: {}", e);
    }
    
    // Update feed status
    self.view_state.feed_errors.remove(&feed_url);
    self.view_state.feed_last_refresh.insert(
        feed_url,
        chrono::Local::now(),
    );
    
    // Reload events for display
    self.load_calendar_events();
}
```

3. Handle FeedError:

5. Add load_calendar_events method:
```rust
fn load_calendar_events(&mut self) {
    let date_str = self.view_state.current_date.format("%Y-%m-%d").to_string();
    match self.db.get_events_for_date(&date_str) {
        Ok(events) => {
            // Separate all-day from timed events
            let (all_day, timed): (Vec<_>, Vec<_>) = events
                .into_iter()
                .partition(|e| e.all_day);
            self.view_state.all_day_events = all_day;
            self.view_state.calendar_events = timed;
        }
        Err(e) => {
            eprintln!("Failed to load calendar events: {}", e);
            self.view_state.calendar_events = Vec::new();
            self.view_state.all_day_events = Vec::new();
        }
    }
}
```
