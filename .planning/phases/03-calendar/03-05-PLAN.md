```rust
use egui::{Color32, Rounding, Stroke, Ui, Vec2};
use crate::calendar::CalendarEvent;

/// Render the all-day events section (above the scrollable timeline)
pub fn render_all_day_events(ui: &mut Ui, events: &[CalendarEvent]) {
    if events.is_empty() {
        return;
    }
    
    ui.horizontal_wrapped(|ui| {
        for event in events {
            let color = parse_color(event.feed_color.as_deref());
            egui::Frame::none()
                .stroke(Stroke::new(2.0, color))
                .inner_margin(4.0)
                .rounding(Rounding::same(4.0))
                .show(ui, |ui| {
                    ui.label(&event.summary);
                });
        }
    });
    ui.add_space(8.0);
    ui.separator();
}

/// Render calendar events for the scrollable area
pub fn render_calendar_events(ui: &mut Ui, events: &[CalendarEvent]) {
    for event in events {
        render_event_card(ui, event);
        ui.add_space(4.0);
    }
    
    if events.is_empty() {
        ui.label(egui::RichText::new("No events").weak().italics());
    }
}

fn render_event_card(ui: &mut Ui, event: &CalendarEvent) {
    let color = parse_color(event.feed_color.as_deref());
    
    egui::Frame::none()
        .fill(ui.visuals().extreme_bg_color)
        .stroke(Stroke::new(3.0, color)) // Left border effect via stroke
        .inner_margin(Vec2::new(8.0, 4.0))
        .rounding(Rounding::same(4.0))
        .show(ui, |ui| {
            ui.horizontal(|ui| {
                ui.label(egui::RichText::new(event.time_display()).weak().small());
                if let Some(name) = &event.feed_name {
                    ui.label(egui::RichText::new(name).weak().small());
                }
            });
            ui.label(&event.summary);
        });
}

fn parse_color(hex: Option<&str>) -> Color32 {
    hex.and_then(|s| {
        let s = s.trim_start_matches('#');
        if s.len() == 6 {
            let r = u8::from_str_radix(&s[0..2], 16).ok()?;
            let g = u8::from_str_radix(&s[2..4], 16).ok()?;
            let b = u8::from_str_radix(&s[4..6], 16).ok()?;
            Some(Color32::from_rgb(r, g, b))
        } else {
            None
        }
    })
    .unwrap_or(Color32::GRAY)
}
```
