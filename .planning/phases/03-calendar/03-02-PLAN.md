use calcard::ical::ICalendar;
use chrono::{Local, NaiveDate, NaiveTime, Duration as ChronoDuration};
use rrule::RRuleSet;

use super::CalendarEvent;

/// Parse iCal data and return events within the specified date range.
/// Recurring events are expanded to individual occurrences.
pub fn parse_ical(
    data: &str,
    feed_url: &str,
    feed_name: Option<&str>,
    feed_color: Option<&str>,
    range_start: NaiveDate,
    range_end: NaiveDate,
) -> Vec<CalendarEvent> {
    let mut events = Vec::new();
    
    let Ok(calendar) = ICalendar::parse(data.as_bytes()) else {
        return events;
    };
    
    for component in calendar.components {
        if let calcard::ical::ICalendarComponent::Event(vevent) = component {
            let uid = vevent.uid.map(|u| u.to_string()).unwrap_or_default();
            let summary = vevent.summary.map(|s| s.to_string()).unwrap_or_default();
            
            // Parse DTSTART - may be date-only (all-day) or datetime
            let Some(dtstart) = vevent.dtstart else { continue };
            
            let (start_date, start_time, all_day) = parse_datetime(&dtstart);
            let Some(start_date) = start_date else { continue };
            
            // Parse DTEND if present
            let (end_date, end_time) = vevent.dtend
                .as_ref()
                .map(|dt| {
                    let (d, t, _) = parse_datetime(dt);
                    (d, t)
                })
                .unwrap_or((None, None));
            
            // Get RRULE if present
            let rrule_str = vevent.rrule.as_ref().map(|r| r.to_string());
            
            // Expand recurring events or create single event
            if let Some(ref rrule) = rrule_str {
                let expanded = expand_rrule(
                    rrule, start_date, start_time, range_start, range_end
                );
                for occurrence_date in expanded {
                    events.push(CalendarEvent {
                        id: None,
                        feed_url: feed_url.to_string(),
                        event_uid: uid.clone(),
                        summary: summary.clone(),
                        dtstart_date: occurrence_date,
                        dtstart_time: start_time,
                        dtend_date: end_date,
                        dtend_time: end_time,
                        all_day,
                        rrule: rrule_str.clone(),
                        feed_name: feed_name.map(String::from),
                        feed_color: feed_color.map(String::from),
                    });
                }
            } else if start_date >= range_start && start_date <= range_end {
                events.push(CalendarEvent {
                    id: None,
                    feed_url: feed_url.to_string(),
                    event_uid: uid,
                    summary,
                    dtstart_date: start_date,
                    dtstart_time: start_time,
                    dtend_date: end_date,
                    dtend_time: end_time,
                    all_day,
                    rrule: None,
                    feed_name: feed_name.map(String::from),
                    feed_color: feed_color.map(String::from),
                });
            }
        }
    }
    
    events
}
